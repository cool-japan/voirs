{{ if .Values.coordinator.enabled }}
To access the coordinator:

  {{- if .Values.ingress.enabled }}
  The coordinator is available at:
  {{- range .Values.ingress.hosts }}
    https://{{ .host }}
  {{- end }}
  {{- else if contains "LoadBalancer" .Values.service.type }}

  NOTE: It may take a few minutes for the LoadBalancer IP to be available.

  Watch the status with:
    kubectl get --namespace {{ .Release.Namespace }} svc {{ include "voirs-evaluation.fullname" . }}-coordinator --watch

  Once available, you can access it at:
    export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "voirs-evaluation.fullname" . }}-coordinator --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
    echo "Coordinator URL: http://$SERVICE_IP:{{ .Values.service.port }}"
  {{- else if contains "ClusterIP" .Values.service.type }}

  Access the coordinator using port-forwarding:
    kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "voirs-evaluation.fullname" . }}-coordinator {{ .Values.service.port }}:{{ .Values.service.port }}
    echo "Coordinator URL: http://127.0.0.1:{{ .Values.service.port }}"
  {{- end }}
{{- end }}
